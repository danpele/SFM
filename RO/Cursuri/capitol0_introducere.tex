\begin{frame}{Obiectul statisticii piețelor financiare}
\itemsize{\small}
\begin{itemize}
    \item \textbf{Statistica piețelor financiare}: metode statistice pentru prețurile, randamentele și riscurile activelor tranzacționate
    \begin{itemize}
        \item descriere: cît de mari și cît de variabile sînt randamentele?
        \item modelare: ce distribuție, ce dependență în timp?
        \item măsurarea riscului: cît de mare poate fi pierderea într-o zi nefavorabilă?
    \end{itemize}
    \item De ce datele financiare cer metode proprii
    \begin{itemize}
        \item zilele extreme sînt mult mai frecvente decît prevede distribuția Normală (\emph{cozi groase})
        \item perioadele calme alternează cu cele agitate (\emph{volatility clustering})
        \item piețele scad împreună în crize: diversificarea slăbește tocmai cînd este cea mai necesară
    \end{itemize}
    \item Aceste regularități sînt \emph{faptele stilizate} ale randamentelor (\refCont), studiate în \sfmchlink{https://danpele.github.io/SFM/RO/Cursuri/capitol2_distributii_clasice_fapte_stilizate.pdf}{Capitolul 2}
\end{itemize}
\end{frame}

\section{De ce piețele financiare?}
%=============================================================================

\begin{frame}{Rolul unei piețe financiare}
\itemsize{\small}
\begin{itemize}
    \item \textbf{Piață financiară}: un loc sau un sistem în care cumpărătorii și vînzătorii schimbă drepturi asupra unor fluxuri de numerar viitoare (acțiuni, obligațiuni, valute, contracte, tokeni), la prețuri stabilite de cerere și ofertă
    \item \textbf{Cinci funcții}
    \begin{itemize}
        \item \textbf{alocarea capitalului}: economiile ajung la firme și la state
        \item \textbf{împărțirea riscului}: riscurile trec la cei dispuși să le poarte
        \item \textbf{lichiditatea}: un activ poate fi transformat repede în bani
        \item \textbf{descoperirea prețului}: prețurile adună informația dispersată între mulți participanți
        \item \textbf{plăți și decontare}: tranzacțiile sînt finalizate în siguranță
    \end{itemize}
    \item Pentru un statistician, piața este un generator de date: fiecare tranzacție lasă în urmă un preț și un moment de timp
\end{itemize}
\end{frame}

\begin{column}{0.47\textwidth}
\begin{itemize}

    \item Investiție \textbf{pasivă} sau \textbf{activă}

\begin{frame}{Formarea prețului: registrul de ordine}
\itemsize{\footnotesize}
\begin{columns}[T]
\begin{column}{0.42\textwidth}
\centering
\begin{tikzpicture}[x=0.5cm, y=0.36cm]
\foreach \p/\q in {1/3, 2/5, 3/4} { \fill[Forest!70] (-\q, -\p) rectangle (0, -\p+0.8); \node[left, font=\tiny] at (-\q, -\p+0.4) {\q}; }
\node[right, font=\tiny] at (0.1, -0.6) {99.99}; \node[right, font=\tiny] at (0.1, -1.6) {99.98}; \node[right, font=\tiny] at (0.1, -2.6) {99.97};
\foreach \p/\q in {1/2, 2/4, 3/6} { \fill[IDAred!70] (-\q, \p) rectangle (0, \p+0.8); \node[left, font=\tiny] at (-\q, \p+0.4) {\q}; }
\node[right, font=\tiny] at (0.1, 1.4) {100.01}; \node[right, font=\tiny] at (0.1, 2.4) {100.02}; \node[right, font=\tiny] at (0.1, 3.4) {100.03};
\draw[dashed, MainBlue] (-7, 0.4) -- (2.5, 0.4);
\node[font=\scriptsize, IDAred] at (-3.5, 5) {oferte de vînzare (ask)};
\node[font=\scriptsize, Forest] at (-3.5, -4.2) {oferte de cumpărare (bid)};
\end{tikzpicture}
\end{column}
\begin{column}{0.55\textwidth}
\begin{itemize}
    \item \textbf{Registrul de ordine}: toate ordinele de cumpărare (\textbf{bid}) și de vînzare (\textbf{ask}) în așteptare, ordonate după preț; barele arată cantitățile
    \item \textbf{Spread-ul bid--ask}: cel mai bun ask minus cel mai bun bid, aici $100{,}01 - 99{,}99 = 0{,}02$, costul unei tranzacții imediate
    \item Un \textbf{ordin la piață} cumpără sau vinde imediat, la cele mai bune prețuri disponibile
    \item \textbf{Întrebări pentru sală}: sosește un ordin la piață de cumpărare a 5 unități
    \begin{itemize}
        \item la ce preț mediu se execută?
        \item cît este spread-ul după aceea?
    \end{itemize}
    \pause
    \item \textbf{Răspuns}
    \begin{itemize}
        \item 2 unități la 100,01 și 3 la 100,02: media $(2 \times 100{,}01 + 3 \times 100{,}02)/5 = 100{,}016$
        \item cel mai bun ask devine 100,02, spread-ul crește la 0,03: ordinul a mișcat prețul
    \end{itemize}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{De ce statistică?}
\itemsize{\small}
\begin{itemize}
    \item \textbf{Prețul de mîine este incert}: îl modelăm ca pe o variabilă aleatoare
    \begin{itemize}
        \item $P_t$: prețul din ziua $t$; șirul $P_1, P_2, \ldots$ este o \textbf{serie de timp}
    \end{itemize}
    \item \textbf{Mărimi studiate în acest curs}
    \begin{itemize}
        \item \textbf{randamentul}: modificarea relativă a prețului (\sfmchlink{https://danpele.github.io/SFM/RO/Cursuri/capitol1_date_randamente_indicatori.pdf}{Capitolul 1})
        \item \textbf{distribuția} randamentelor și cît de groase îi sînt cozile (\sfmchlink{https://danpele.github.io/SFM/RO/Cursuri/capitol2_distributii_clasice_fapte_stilizate.pdf}{Capitolele 2}, \sfmchlink{https://danpele.github.io/SFM/RO/Cursuri/capitol3_distributii_alfa_stabile.pdf}{3} și \sfmchlink{https://danpele.github.io/SFM/RO/Cursuri/capitol5_cozi_groase_evt.pdf}{5})
        \item \textbf{volatilitatea}: abaterea standard a randamentelor și felul în care se schimbă în timp (Capitolele 8 și 9)
        \item \textbf{măsuri de risc}: VaR (Value at Risk, valoarea expusă la risc) și ES (Expected Shortfall) (Capitolul 10)
    \end{itemize}
    \item \textbf{Marea întrebare}: pot prețurile trecute prezice randamentele viitoare?
    \begin{itemize}
        \item potrivit ipotezei pieței eficiente, aproape deloc (\sfmchlink{https://danpele.github.io/SFM/RO/Cursuri/capitol7_piete_eficiente_mers_aleator_vr.pdf}{Capitolul 7})
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Recapitulare: de ce piețele financiare?}
\itemsize{\small}
\begin{itemize}
    \item Piețele alocă capitalul, împart riscul, asigură lichiditatea și descoperă prețurile
    \item Prețurile se formează în registrul de ordine: spread-ul este costul unei tranzacții imediate
    \item Șase clase de active, un singur set de instrumente statistice
    \item Prețul de mîine este o variabilă aleatoare: statistica îi descrie randamentele, volatilitatea și riscul
\end{itemize}
\end{frame}

\section{De la prețuri la randamente: primii pași}
%=============================================================================

\begin{frame}{Randamente simple și logaritmice}
\itemsize{\small}
\begin{itemize}
    \item $P_t$: prețul de la sfîrșitul zilei $t$ (pentru ETF-uri și acțiuni, prețul ajustat pentru dividende și splituri)
    \item \textbf{Randament simplu}: $R_t = \dfrac{P_t}{P_{t-1}} - 1$
    \begin{itemize}
        \item cîștigul procentual al unei zile: ce primește investitorul
    \end{itemize}
    \item \textbf{Randament logaritmic}: $r_t = \ln \dfrac{P_t}{P_{t-1}} = \ln(1 + R_t)$
    \begin{itemize}
        \item aproape egal cu $R_t$ pentru variații mici; mai mic pentru variații mari
    \end{itemize}
    \item \textbf{Exemplu}: $P_0 = 100$, $P_1 = 110$, $P_2 = 99$
    \begin{itemize}
        \item $R_1 = +10\%$, $r_1 = \ln 1{,}1 = +$9{,}53\%
        \item $R_2 = -10\%$, $r_2 = \ln 0{,}9 = $ -10{,}54\%
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Ce randamente se adună?}
\itemsize{\footnotesize}
\begin{itemize}
    \item \textbf{Întrebări pentru sală}: același exemplu, $100 \to 110 \to 99$
    \begin{itemize}
        \item cît este suma celor două randamente simple?
        \item a cîștigat, a pierdut sau a rămas pe loc investitorul în cele două zile?
    \end{itemize}
    \pause
    \item \textbf{Răspuns}
    \begin{itemize}
        \item $R_1 + R_2 = 10\% - 10\% = 0\%$, dar investitorul a pierdut: $99/100 - 1 = -1\%$
        \item randamentele simple \textbf{se compun}: $(1 + R_1)(1 + R_2) - 1 = 1{,}1 \times 0{,}9 - 1 = -1\%$
        \item randamentele logaritmice \textbf{se adună}: $r_1 + r_2 = $ -1{,}01\% $= \ln(99/100)$
    \end{itemize}
    \item Regula: în timp, adunăm randamentele logaritmice și compunem randamentele simple (detalii în \sfmchlink{https://danpele.github.io/SFM/RO/Cursuri/capitol1_date_randamente_indicatori.pdf}{Capitolul 1})
\end{itemize}
\end{frame}

\begin{frame}{Media, volatilitatea și anualizarea}
\itemsize{\footnotesize}
\begin{itemize}
    \item Din $n$ randamente logaritmice zilnice $r_1, \ldots, r_n$
    \begin{itemize}
        \item \textbf{media}: $\bar r = \frac{1}{n} \sum_{t=1}^n r_t$
        \item \textbf{volatilitatea}: abaterea standard $s = \sqrt{\frac{1}{n-1} \sum_{t=1}^n (r_t - \bar r)^2}$
    \end{itemize}
    \item \textbf{Anualizarea} cu $q$ observații pe an
    \begin{itemize}
        \item media: $q \bar r$; volatilitatea: $\sqrt{q}\, s$, pentru că varianțele zilelor independente se adună
        \item $q \approx 252$ pentru burse, $q = 365$ pentru Bitcoin
    \end{itemize}
    \item \textbf{Exemplu}
    \begin{itemize}
        \item un indice bursier cu volatilitate zilnică de 1\%: $\sqrt{252} \times 1\% = $ 15{,}9\% pe an
        \item Bitcoin cu volatilitate zilnică de 3\%: $\sqrt{365} \times 3\% = $ 57{,}3\% (cu 252, greșit: 47{,}6\%)
    \end{itemize}
\end{itemize}
\end{frame}

    \begin{itemize}

    \end{itemize}

\begin{column}{0.58\textwidth}
\begin{itemize}
    \item \textbf{Vîrful curent}: $M_t = \max_{s \le t} P_s$, cel mai mare preț de pînă acum
    \item \textbf{Drawdown}: $DD_t = P_t / M_t - 1$, niciodată pozitiv
    \item \textbf{Drawdown-ul maxim} (MDD, maximum drawdown): cea mai mică valoare a lui $DD_t$; aici $-25\%$, de la 120 la 90
    \item \textbf{De ce îl urmăresc investitorii}
    \begin{itemize}
        \item este pierderea celui care a cumpărat în cel mai nefavorabil moment
        \item nu spune cît durează revenirea: raportați și durata
    \end{itemize}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Recapitulare: de la prețuri la randamente}
\itemsize{\small}
\begin{itemize}
    \item $R_t = P_t/P_{t-1} - 1$ se compune; $r_t = \ln(P_t/P_{t-1})$ se adună în timp
    \item Anualizăm media cu $q$ și volatilitatea cu $\sqrt{q}$; $q = 365$ pentru Bitcoin
    \item Randamentele zilnice: volatility clustering și cozi groase (coeficient de aplatizare 13{,}7, față de 3)
    \item Drawdown: pierderea față de vîrful anterior; drawdown-ul maxim rezumă cel mai sever episod
\end{itemize}
\end{frame}

\begin{frame}{O întrebare deschisă}
\itemsize{\small}
\begin{itemize}
    \item \textbf{Întrebarea}: a devenit Bitcoin mai puțin volatil de cînd a intrat în finanțele tradiționale?
    \item \textbf{De ce contează}
    \begin{itemize}
        \item un Bitcoin mai puțin volatil ar fi mai ușor de ținut într-un fond de pensii sau în portofoliul unei bănci
        \item ETF-urile spot din ianuarie 2024 au deschis Bitcoin investitorilor care cumpără prin conturi obișnuite de brokeraj (\refSEC)
    \end{itemize}
    \item \textbf{De ce este deschisă}
    \begin{itemize}
        \item doar cîțiva ani de date după 2024
        \item există alte explicații: o piață mai calmă în ansamblu, o piață mai mare și mai lichidă
        \item rezultate conexe: activele cripto și cele tradiționale se apropie în timp (\refPeleCrypto)
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Contribuția posibilă a AI}
\itemsize{\small}
\begin{itemize}
    \item \textbf{Literatura}: cereți articole din reviste cu evaluare colegială (peer review) despre volatilitatea Bitcoin, cu DOI
    \begin{itemize}
        \item apoi deschideți voi fiecare DOI: asistenții AI inventează referințe plauzibile
    \end{itemize}
    \item \textbf{Ipoteze}: cereți trei explicații concurente pentru o scădere a volatilității
    \begin{itemize}
        \item fiecare explicație trebuie să prezică ceva ce celelalte nu prezic
    \end{itemize}
    \item \textbf{Codul}: cereți o primă variantă care calculează volatilitatea pentru fiecare an
    \begin{itemize}
        \item verificați anualizarea, calendarul și datele, linie cu linie
    \end{itemize}
    \item \textbf{Critica}: cereți asistentului să joace rolul unui recenzent ostil al rezultatului vostru
    \begin{itemize}
        \item învățarea automată poate genera ipoteze; testarea lor rămîne sarcina cercetătorului (\refLM)
    \end{itemize}
\end{itemize}
\end{frame}
